  <script>

    let players = JSON.parse(localStorage.getItem('pe_tourney_players')) || [

      { id: 1, name: "Lucas M.", points: 7.5, wins: 3, losses: 1, refs: 2 },

      { id: 2, name: "Emma R.", points: 6.0, wins: 3, losses: 1, refs: 0 },

      { id: 3, name: "Noah T.", points: 5.5, wins: 2, losses: 2, refs: 1 },

      { id: 4, name: "Sophia K.", points: 4.0, wins: 1, losses: 2, refs: 2 },

      { id: 5, name: "Oliver B.", points: 3.5, wins: 1, losses: 1, refs: 1 },

      { id: 6, name: "Mia D.", points: 2.0, wins: 0, losses: 3, refs: 4 }

    ];


    function persist() {

      localStorage.setItem('pe_tourney_players', JSON.stringify(players));

      render();

    }


    function switchTab(tab) {

      ['standings', 'match', 'roster'].forEach(t => {

        document.getElementById('view-' + t).classList.add('hidden');

        document.getElementById('btn-' + t).classList.remove('active');

      });

      document.getElementById('view-' + tab).classList.remove('hidden');

      document.getElementById('btn-' + tab).classList.add('active');


      if (tab === 'match') setupMatchOptions();

    }


    function render() {

      players.sort((a, b) => b.points - a.points);

      const list = document.getElementById('player-list');

      list.innerHTML = players.map((p, idx) => `

        <div class="card">

          <div class="row">

            <div style="display: flex; align-items: center; gap: 0.5rem;">

              <span class="rank-num ${idx < 3 ? 'top-rank' : ''}">#${idx + 1}</span>

              <div>

                <div class="player-name">${p.name}</div>

                <div class="meta-text">Refereed: ${p.refs} games</div>

              </div>

            </div>

            <div style="text-align: right;">

              <div class="score-badge">${p.points.toFixed(1)} pts</div>

              <div class="meta-text">${p.wins}W - ${p.losses}L</div>

            </div>

          </div>

        </div>

      `).join('');

    }


    function setupMatchOptions() {

      players.sort((a, b) => b.points - a.points);

      const challengerSelect = document.getElementById('match-challenger');

      const refSelect = document.getElementById('match-referee');


      challengerSelect.innerHTML = players.map((p, i) => `<option value="${p.id}">${p.name} (#${i + 1})</option>`).join('');

      refSelect.innerHTML = '<option value="">-- No Referee --</option>' + players.map(p => `<option value="${p.id}">${p.name}</option>`).join('');

      filterOpponents();

    }


    function filterOpponents() {

      players.sort((a, b) => b.points - a.points);

      const cId = parseInt(document.getElementById('match-challenger').value);

      const cIdx = players.findIndex(p => p.id === cId);

      const minIdx = Math.max(0, cIdx - 5);


      const eligible = players.filter((p, i) => i >= minIdx && p.id !== cId);

      document.getElementById('match-opponent').innerHTML = eligible.map(p => {

        const pIdx = players.indexOf(p) + 1;

        return `<option value="${p.id}">${p.name} (#${pIdx})</option>`;

      }).join('');

    }


    function recordMatch() {

      const cId = parseInt(document.getElementById('match-challenger').value);

      const oId = parseInt(document.getElementById('match-opponent').value);

      const rId = parseInt(document.getElementById('match-referee').value);

      const winner = document.getElementById('match-winner').value;


      if (!cId || !oId || cId === oId) return alert('Select two distinct players.');


      const cIdx = players.findIndex(p => p.id === cId);

      const oIdx = players.findIndex(p => p.id === oId);

      const challenger = players[cIdx];

      const opponent = players[oIdx];

      const rankDiff = Math.max(0, cIdx - oIdx);


      if (winner === 'challenger') {

        challenger.points += (1.0 + (rankDiff * 0.5));

        challenger.wins++;

        opponent.losses++;

      } else {

        opponent.points += 1.0;

        opponent.wins++;

        challenger.points += 0.5;

        challenger.losses++;

      }


      if (rId) {

        const referee = players.find(p => p.id === rId);

        if (referee && referee.id !== cId && referee.id !== oId) {

          referee.points += 0.5;

          referee.refs++;

        }

      }


      persist();

      switchTab('standings');

    }


    function importStudents() {

      const val = document.getElementById('roster-input').value.trim();

      if (!val) return;

      const names = val.split('\n').map(n => n.trim()).filter(n => n.length > 0);

      players = names.map((name, i) => ({ id: Date.now() + i, name, points: 0.0, wins: 0, losses: 0, refs: 0 }));

      document.getElementById('roster-input').value = '';

      persist();

      switchTab('standings');

    }


    function resetData() {

      if (confirm('Clear all player data and reset scores?')) {

        players = [];

        persist();

      }

    }


    render();
